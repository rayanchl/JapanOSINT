	const url = "https://data.earth.jaxa.jp/stac/cog/v1/catalog.json";

	const c = new je1.stac.Catalog({ url });
	await c.init();

	const collection = c.getChildUrlAll().find(e => {
		return e.indexOf("/" + id + "/collection.json") > 0;
	});

	if (collection == undefined) {
		throw new Error();
	}

	return collection;
}
